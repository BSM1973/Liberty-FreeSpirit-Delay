cmake_minimum_required(VERSION 3.22)
project(FreeSpiritDelay VERSION 0.1.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FSD_CORE_ONLY "Build DSP tests without JUCE" OFF)
enable_testing()
add_executable(FreeSpiritCoreTests Tests/CoreTests.cpp)
target_include_directories(FreeSpiritCoreTests PRIVATE Source)
add_test(NAME FreeSpiritCoreTests COMMAND FreeSpiritCoreTests)
if(FSD_CORE_ONLY)
 return()
endif()
if(DEFINED JUCE_DIR AND EXISTS "${JUCE_DIR}/CMakeLists.txt")
 add_subdirectory("${JUCE_DIR}" "${CMAKE_BINARY_DIR}/JUCE")
else()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.8 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
endif()
set(FSD_FORMATS VST3 Standalone)
if(APPLE)
 list(APPEND FSD_FORMATS AU)
endif()
juce_add_plugin(FreeSpiritDelay COMPANY_NAME "Liberty Music Software" COMPANY_WEBSITE "https://libertymusicsoftware.com" BUNDLE_ID "com.libertymusicsoftware.freespiritdelay" PLUGIN_MANUFACTURER_CODE Lbms PLUGIN_CODE Fsdl FORMATS ${FSD_FORMATS} PRODUCT_NAME "FreeSpirit Delay" IS_SYNTH FALSE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_generate_juce_header(FreeSpiritDelay)
target_sources(FreeSpiritDelay PRIVATE Source/Plugin.cpp Source/Editor.cpp Source/Presets.cpp)
target_compile_definitions(FreeSpiritDelay PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(FreeSpiritDelay PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_gui_extra juce::juce_recommended_config_flags)
if(MSVC)
 target_compile_options(FreeSpiritDelay PRIVATE /utf-8)
endif()
